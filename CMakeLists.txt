cmake_minimum_required(VERSION 3.16)
project(libdrda VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

option(DRDA_WERROR "Treat warnings as errors" OFF)
option(DRDA_BUILD_TOOLS "Build drdacli" ON)
option(DRDA_BUILD_TESTS "Build the tests" ON)
option(DRDA_WITH_OPENSSL "TLS through OpenSSL" ON)

if(MSVC)
  # getenv, sscanf and friends are standard C; MSVC's "use the _s variant"
  # deprecations would otherwise fail the -Werror build.
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
  set(DRDA_WARN /W4)
  if(DRDA_WERROR)
    list(APPEND DRDA_WARN /WX)
  endif()
else()
  set(DRDA_WARN -Wall -Wextra -Wshadow -Wconversion -Wno-sign-conversion)
  if(DRDA_WERROR)
    list(APPEND DRDA_WARN -Werror)
  endif()
endif()

add_library(drda STATIC src/wire.c src/decode.c src/drda.c)
target_include_directories(drda PUBLIC include PRIVATE src)
target_compile_options(drda PRIVATE ${DRDA_WARN})
if(DRDA_WITH_OPENSSL)
  find_package(OpenSSL 1.1.1 REQUIRED)
  target_link_libraries(drda PUBLIC OpenSSL::SSL OpenSSL::Crypto)
  target_compile_definitions(drda PRIVATE DRDA_HAVE_OPENSSL)
  if(WIN32)
    target_link_libraries(drda PUBLIC crypt32)
  endif()
endif()
if(WIN32)
  target_link_libraries(drda PUBLIC ws2_32)
  target_compile_definitions(drda PRIVATE _CRT_SECURE_NO_WARNINGS WIN32_LEAN_AND_MEAN)
endif()

if(DRDA_BUILD_TOOLS)
  add_executable(drdacli tools/drdacli.c)
  target_link_libraries(drdacli PRIVATE drda)
  target_compile_options(drdacli PRIVATE ${DRDA_WARN})
  if(WIN32)
    target_link_libraries(drdacli PRIVATE shell32)
  endif()
endif()

if(DRDA_BUILD_TESTS)
  enable_testing()
  add_executable(test_decode tests/test_decode.c src/wire.c src/decode.c)
  target_include_directories(test_decode PRIVATE src)
  target_compile_options(test_decode PRIVATE ${DRDA_WARN})
  add_test(NAME decode COMMAND test_decode)

  # Runs against a live server when DRDA_TEST_HOST is set; skips otherwise.
  find_package(Threads REQUIRED)
  add_executable(test_live tests/test_live.c)
  target_link_libraries(test_live PRIVATE drda Threads::Threads)
  target_compile_options(test_live PRIVATE ${DRDA_WARN})
  add_test(NAME live COMMAND test_live)
  set_tests_properties(live PROPERTIES SKIP_RETURN_CODE 77)
endif()
